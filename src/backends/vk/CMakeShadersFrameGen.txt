set(FRAMEGEN_BASE_ARGS
    -reflection -deps=gcc -DFFX_GPU=1 -DFFXM_GPU=1)

get_filename_component(FFXM_TOOLS_BIN_DIR "${FFXM_SC_EXECUTABLE}" DIRECTORY)

file(RELATIVE_PATH FRAMEGEN_REL_GLSLANG "${CMAKE_SOURCE_DIR}" "${FFXM_TOOLS_BIN_DIR}/glslangValidator.exe")
file(RELATIVE_PATH FRAMEGEN_REL_GPU_PATH "${CMAKE_SOURCE_DIR}" "${FFXM_GPU_PATH}")
file(RELATIVE_PATH FRAMEGEN_REL_OUTPUT "${CMAKE_SOURCE_DIR}" "${FFXM_PASS_SHADER_OUTPUT_PATH}")

set(FRAMEGEN_GLSL_BASE_ARGS
    -compiler=glslang -glslangexe=${FRAMEGEN_REL_GLSLANG} -V -S comp --target-env vulkan1.1 -Os -DFFX_GLSL=1 -DFFXM_GLSL=1)

# 1. Optical Flow Prepare Pass
set(OF_PREPARE_SHADER "${CMAKE_CURRENT_SOURCE_DIR}/shaders/opticalflow/ffx_opticalflow_prepare_pass.glsl")
set(OF_PREPARE_PERMUTATION_HEADER "${FFXM_PASS_SHADER_OUTPUT_PATH}/ffxm_opticalflow_prepare_pass_permutations.h")

if(EXISTS "${OF_PREPARE_SHADER}")
    file(RELATIVE_PATH OF_REL_SHADER_PREPARE "${CMAKE_SOURCE_DIR}" "${OF_PREPARE_SHADER}")
    add_custom_command(
        OUTPUT ${OF_PREPARE_PERMUTATION_HEADER}
        COMMAND ${FFXM_SC_EXECUTABLE}
            ${FRAMEGEN_BASE_ARGS}
            ${FRAMEGEN_GLSL_BASE_ARGS}
            "-I${FRAMEGEN_REL_GPU_PATH}"
            -name=ffxm_opticalflow_prepare_pass
            -output=${FRAMEGEN_REL_OUTPUT}
            ${OF_REL_SHADER_PREPARE}
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        DEPENDS ${OF_PREPARE_SHADER}
        COMMENT "Compiling Optical Flow Prepare shader permutations with FidelityFX-SC..."
        VERBATIM
    )
    add_shader_output("${OF_PREPARE_PERMUTATION_HEADER}")
endif()
